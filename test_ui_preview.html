Mã Nguồn Widget 4x4 Cho iPhone

Dán vào app Scriptable (miễn phí trên App Store)

// Scriptable Widget 4x4: Couple Locket & Habi
// 1. Tải app "Scriptable" trên App Store (Free)
// 2. Tạo Script mới, dán toàn bộ đoạn code này vào
// 3. Ra màn hình chính iPhone > Thêm Widget Scriptable kích thước Lớn (4x4)

const COUPLE_API = "https://your-couple-app.web.app/api/widget.json";

async function createWidget() {
  const widget = new ListWidget();
  widget.backgroundColor = new Color("#0f172a");

  // Lấy dữ liệu mới nhất
  const req = new Request(COUPLE_API);
  const data = await req.loadJSON();

  // Tải ảnh Locket
  const imgReq = new Request(data.photoUrl);
  const image = await imgReq.loadImage();
  
  widget.backgroundImage = image;

  // Overlay caption & mood
  const footer = widget.addStack();
  footer.layoutVertically();
  footer.bottomAlignContent();
  
  const moodText = footer.addText(data.moodEmoji + " " + data.caption);
  moodText.font = Font.boldSystemFont(14);
  moodText.textColor = Color.white();
  moodText.shadowColor = Color.black();
  moodText.shadowRadius = 4;

  const daysText = footer.addText("❤️ Bên nhau " + data.daysTogether + " ngày");
  daysText.font = Font.systemFont(11);
  daysText.textColor = new Color("#fecdd3");

  return widget;
}

const widget = await createWidget();
Script.setWidget(widget);
Script.complete();
